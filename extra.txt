public:
    __device__ camera(
        vec lookfrom,
        vec lookat,
        vec vup,
        float vfov, // Vertical Field of View in degrees
        float aspect_ratio,
        float aperture,
        float focus_dist
    ) {
        // 1. Calculate Field of View
        float theta = vfov * (3.14159265f / 180.0f); // Convert degrees to radians
        float h = tan(theta / 2.0f);
        float viewport_height = 2.0f * h;
        float viewport_width = aspect_ratio * viewport_height;

        // 2. Calculate the camera's local rotation axes
        w = unit_vector(lookfrom - lookat);
        u = unit_vector(cross(vup, w));
        v = cross(w, u);

        // 3. Set the camera physical properties
        origin = lookfrom;
        horizontal = focus_dist * viewport_width * u;
        vertical = focus_dist * viewport_height * v;
        lower_left_corner = origin - horizontal / 2.0f - vertical / 2.0f - focus_dist * w;

        // 4. Set the lens size
        lens_radius = aperture / 2.0f;
    }

    // NEW: We pass local_state to generate the random blur!
    __device__ ray get_ray(float s, float t, curandState* local_state) const {
        vec rd = lens_radius * random_in_unit_disk(local_state);
        vec offset = u * rd.x() + v * rd.y();

        return ray(
            origin + offset, 
            lower_left_corner + s * horizontal + t * vertical - origin - offset
        );
    }
